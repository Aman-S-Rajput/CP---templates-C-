#include <bits/stdc++.h>
using namespace std;
// also solution for line segment trace 1
const long long INF = 2e18; // Large negative infinity for max queries

// Represents y = m * x + c
struct Line {
    long long m, c;
    Line() : m(0), c(-INF) {}
    Line(long long m, long long c) : m(m), c(c) {}
    long long eval(long long x) {
        return m * x + c;
    }
};

const int MAX_M = 1000005; // Assuming max M is up to 10^6
Line tree_lc[4 * MAX_M];

// Standard Li Chao Tree insertion
void insert_line(int node, int l, int r, Line line) {
    if (l == r) {
        if (line.eval(l) > tree_lc[node].eval(l)) {
            tree_lc[node] = line;
        }
        return;
    }

    int mid = l + (r - l) / 2;

    // Check which line is better at the boundaries and midpoint
    bool left_better = line.eval(l) > tree_lc[node].eval(l);
    bool mid_better = line.eval(mid) > tree_lc[node].eval(mid);

    // The line that is better at the midpoint becomes the new dominant line for this node
    if (mid_better) {
        swap(tree_lc[node], line);
    }

    // The "loser" line gets pushed down to the half where it could still be optimal
    if (left_better != mid_better) {
        insert_line(2 * node, l, mid, line);
    } else {
        insert_line(2 * node + 1, mid + 1, r, line);
    }
}

// Query the maximum value at a specific x-coordinate
long long query(int node, int l, int r, int x) {
    long long ans = tree_lc[node].eval(x);
    if (l == r) return ans;

    int mid = l + (r - l) / 2;
    if (x <= mid) {
        ans = max(ans, query(2 * node, l, mid, x));
    } else {
        ans = max(ans, query(2 * node + 1, mid + 1, r, x));
    }
    return ans;
}

int main() {
    // Fast I/O
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n;
    long long m_max;
    if (!(cin >> n >> m_max)) return 0;

    // Read all n lines and insert them into the Li Chao Tree
    for (int i = 0; i < n; i++) {
        long long y1, y2;
        cin >> y1 >> y2;
        long long slope = (y2 - y1) / m_max;
        long long intercept = y1;
        insert_line(1, 0, m_max, Line(slope, intercept));
    }

    // Query every integer coordinate from 0 to m
    for (long long x = 0; x <= m_max; x++) {
        cout << query(1, 0, m_max, x) << (x == m_max ? "" : " ");
    }
    cout << "\n";

    return 0;
}
