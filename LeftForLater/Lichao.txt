const long long INF = 2e18; // Use -INF for max queries, +INF for min queries

struct Line {
    long long m, c;

    Line() : m(0), c(-INF) {} // Default loser line for MAX queries
    Line(long long m, long long c) : m(m), c(c) {}

    long long eval(long long x) {
        return m * x + c;
    }
};

struct LiChaoTree {
    long long min_x, max_x;
    vector<Line> tree;

    // Generic constructor taking explicit left and right boundaries
    LiChaoTree(long long min_x, long long max_x) {
        this->min_x = min_x;
        this->max_x = max_x;

        long long domain_size = max_x - min_x + 1;
        tree.assign(4 * domain_size + 5, Line());
    }

    void _insert_line(int node, long long l, long long r, Line new_line) {
        if (l == r) {
            if (new_line.eval(l) > tree[node].eval(l)) {
                tree[node] = new_line;
            }
            return;
        }

        long long mid = l + (r - l) / 2;

        bool left_better = new_line.eval(l) > tree[node].eval(l);
        bool mid_better = new_line.eval(mid) > tree[node].eval(mid);

        if (mid_better) {
            swap(tree[node], new_line);
        }

        if (left_better != mid_better) {
            _insert_line(2 * node, l, mid, new_line);
        } else {
            _insert_line(2 * node + 1, mid + 1, r, new_line);
        }
    }

    void _add_segment(int node, long long l, long long r, long long ql, long long qr, Line new_line) {
        if (ql > r || qr < l) return;

        if (ql <= l && r <= qr) {
            _insert_line(node, l, r, new_line);
            return;
        }

        long long mid = l + (r - l) / 2;
        _add_segment(2 * node, l, mid, ql, qr, new_line);
        _add_segment(2 * node + 1, mid + 1, r, ql, qr, new_line);
    }

    long long _query(int node, long long l, long long r, long long x) {
        long long ans = tree[node].eval(x);
        if (l == r) return ans;

        long long mid = l + (r - l) / 2;

        if (x <= mid) {
            ans = max(ans, _query(2 * node, l, mid, x));
        } else {
            ans = max(ans, _query(2 * node + 1, mid + 1, r, x));
        }
        return ans;
    }

    // --- PUBLIC API ---

    void add_global_line(long long m, long long c) {
        _insert_line(1, min_x, max_x, Line(m, c));
    }

    void add_line_segment(long long m, long long c, long long start_x, long long end_x) {
        // Clamp the segment to the tree's valid domain to prevent out-of-bounds traversal
        start_x = max(start_x, min_x);
        end_x = min(end_x, max_x);

        if (start_x > end_x) return; // Completely outside the domain

        _add_segment(1, min_x, max_x, start_x, end_x, Line(m, c));
    }

    long long get_max(long long x) {
        // Safety check to prevent traversing outside the segment tree's domain
        if (x < min_x || x > max_x) return -INF;

        return _query(1, min_x, max_x, x);
    }
};
