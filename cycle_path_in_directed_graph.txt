const int N = 1e5 + 5;
std::vector<int> adj[N], par(N), col(N);
int cs = -1, ce = -1;
bool dfs(int v)
{
    col[v] = 1;
    for(int u : adj[v])
    {
        if(col[u] == 0){
            par[u] = v;
            if(dfs(u))return 1;
        }
        else if(col[u] == 1){
            ce = v, cs =u;
            return 1;
        }
    }
    col[v] = 2;
    return 0;

}
void solve()
{
    int n, m;
    cin>>n>>m;
    forn(i,m)
    {
        int a, b;
        cin>>a>>b;
        --a,--b;
        adj[a].push_back(b);
    }
    forn(i, n)
    {
        if(col[i] == 0 and dfs(i))
            break;
    }
    if(ce == -1){
        cout<<"IMPOSSIBLE\n";
    }
    else{
        vector<int>cycle;
        cycle.push_back(cs);
        for(int v = ce; v!= cs ; v = par[v])
            cycle.push_back(v);
        cycle.push_back(cs);
        reverse(all(cycle));
        cout<<cycle.size()<<"\n";
        for(auto &x: cycle)cout<<x + 1<<' ';
    }
}
