const int N = 2e5 + 5;
std::vector<int> par(N), siz(N, 1), add_on(N, inf);
int find(int i)
{
    while(i != par[i])i = par[i];
    return i;
}
void merge(int a,int b,int day)
{
    a = find(a), b = find(b);
    if(a != b)
    {
        if(siz[a] < siz[b])swap(a, b); // a is bigger
        par[b] = a;
        siz[a] += siz[b];
        add_on[b] = day;
    }
}
int find_merge(int a, int b)
{
    int ans = 0;
    while(a != b)
    {
        if(add_on[a] > add_on[b])swap(a, b);// a merges earlier
        ans = add_on[a];
        if(add_on[a] == inf)return -1;
        a = par[a];
    }
    return ans;
}
void solve()
{
    int n, m, q;
    cin>>n>>m>>q;
    for(int i = 1;i<= n;i++)par[i] = i;
    for(int day = 1;day <= m;day++)
    {
        int a, b;cin>>a>>b;
        merge(a, b,day);
    }
    while(q--)
    {
        int a, b;cin>>a>>b;
        cout<<find_merge(a, b)<<nl;
    }

}

signed main()
{
    ios_base::sync_with_stdio(false), cin.tie(NULL);
    int t = 1;
    // cin>>t;
    while(t--)solve();
}
