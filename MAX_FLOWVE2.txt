const int N = 505;
std::vector<int> adj[N], parent(N,-1);
vector<vector<int>>capacity(N, vector<int>(N));
int n, m;
bool reachable(){
    queue<int>q;
    q.push(1);
    while(!q.empty()){
        int node = q.front();
        q.pop();
        for(auto &son : adj[node]){
            int cap = capacity[node][son];
            if(cap <= 0 || parent[son]!= -1)continue;
            parent[son] = node;
            q.push(son);
        }
    }
    return parent[n] != -1;
}

int max_flow(){
    int maxF = 0;
    parent[1] = 1;// slightly optimizes depending on the graph
    while(reachable()){
        int node = n,curr_flow = LLONG_MAX;
        while(node != 1){
            curr_flow = min(curr_flow, capacity[parent[node]][node]);
            node = parent[node];
        }
        node = n;
        while(node != 1){
            capacity[parent[node]][node] -= curr_flow;
            capacity[node][parent[node]] += curr_flow; /// to block backedges(allows algo to undo flow sent in wrong direction)
            node = parent[node];
        }
        maxF += curr_flow;
        fill(parent.begin(),parent.begin() + n + 1, -1);
    }
    return maxF;
}

void solve()
{
    cin>>n>>m;
    for(int i = 0;i < m;i++){
        int a, b, c;
        cin>>a>>b>>c;
        adj[a].push_back(b);
        adj[b].push_back(a);
        capacity[a][b] += c;
    }
    cout<<max_flow();
}

    while(t--)solve();
