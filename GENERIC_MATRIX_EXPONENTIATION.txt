const int mod = 1e9 + 7;
struct Matrix{
    int r, c;
    vector<vector<int>> mat;
    Matrix(int row, int col){
        r = row;
        c = col;
        mat.resize(r, vector<int>(c));
    }
    Matrix operator*(const Matrix& other) const{
        Matrix result(r, other.c);
        for(int i = 0;i < r;i++){
            for(int j = 0;j < other.c;j++){
                for(int k = 0;k < c;k++){
                    result.mat[i][j] += mat[i][k] * other.mat[k][j];
                    result.mat[i][j]%= mod;
                }
            }
        }
        return result;
    }
};
